function kmTo(b) {
  if (!S.pos) return null;
  var R = 6371, dLat = (b.lat - S.pos.lat) * Math.PI / 180, dLng = (b.lng - S.pos.lng) * Math.PI / 180;
  var a = Math.sin(dLat / 2) * Math.sin(dLat / 2) + Math.cos(S.pos.lat * Math.PI / 180) * Math.cos(b.lat * Math.PI / 180) * Math.sin(dLng / 2) * Math.sin(dLng / 2);
  return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
}

var WRAP = false;

function startCam() {
  stopCam();
  var v = $('#cam'), msg = $('#camMsg');
  if (WRAP) { if (S.camAsked || S.busy) return; S.camAsked = true; try { window.top.postMessage({ tc: 'scan', lang: S.lang }, '*'); return; } catch (e) {} }

  navigator.mediaDevices.getUserMedia({ video: { facingMode: 'environment' } }).then(function (st) {
    camStream = st; v.srcObject = st; v.hidden = false; v.play();

  }).catch(function () { msg.innerHTML = '<div class="alert warn">' + T('noCam') + '</div>'; });

function pollCharge(now) {
  clearInterval(chargePoll);
  var tick = function () {
    if (S.view !== 'charge' && S.view !== 'topup') { clearInterval(chargePoll); return; }
    call('charge.status').then(function (d) {
      var was = S.charge || {};
      if (d.active) {
        if (d.near && !was.near) { toast(T('nearAlert')); buzz(); }
        if (d.low && !was.low) { toast(T('lowAlert')); buzz(); }
        S.charge = d; S.me.balance = d.balance; S.me.charging = d.id;
      } else {
        clearInterval(chargePoll); S.charge = null; S.me.charging = ''; S.me.balance = d.balance; S.hist = null;

      }
      if (S.view === 'charge') render(true);
    }).catch(function () {});
  };
  if (now) tick();
  chargePoll = setInterval(tick, 5000);
}

/** phone photos are big: resize to 1280px JPEG before sending */
function shrink(file, cb) {
  var r = new FileReader();
  r.onload = function () { var img = new Image(); img.onload = function () { var s = Math.min(1, 1280 / Math.max(img.width, img.height)), c = document.createElement('canvas'); c.width = img.width * s; c.height = img.height * s; c.getContext('2d').drawImage(img, 0, 0, c.width, c.height); cb(c.toDataURL('image/jpeg', 0.72)); }; img.src = r.result; };
  r.readAsDataURL(file);
}
document.addEventListener('submit', function (e) {
  e.preventDefault();
  var id = e.target.id, fm = S.form;
  if (id === 'authForm') {
    S.err = '';
    if (S.authMode === 'register') {
      if ((fm.name || '').trim().length < 2) S.err = T('e_name');
      else if (String(fm.phone || '').replace(/\D/g, '').length < 8) S.err = T('e_phone');
      else if ((fm.pass || '').length < 6) S.err = T('e_passShort');
      else if (fm.pass !== fm.pass2) S.err = T('e_pass2');
      if (S.err) { render(true); return; }
      S.busy = T('creating'); render(true);
      call('auth.register', { name: fm.name, phone: fm.phone, pass: fm.pass, lang: S.lang }).then(signedIn).catch(authFail);
    } else {
      if (!fm.id || !fm.pass) { S.err = T('e_none'); render(true); return; }
      S.busy = T('signingIn'); render(true);
      call('auth.login', { id: fm.id, pass: fm.pass }).then(signedIn).catch(authFail);
    }
  }
  if (id === 'profForm') { S.busy = T('save'); render(true); call('me.update', { name: fm.name, phone: fm.phone }).then(function (me) { S.busy = ''; S.me = me; S.err = ''; toast(T('saved')); ACT.backMe(); }).catch(function (er) { S.busy = ''; S.err = errText(er); render(true); }); }
  if (id === 'pwForm') {
    if ((fm.pass || '').length < 6) { S.err = T('e_passShort'); render(true); return; } if (fm.pass !== fm.pass2) { S.err = T('e_pass2'); render(true); return; }
    S.busy = T('save'); render(true);
    call('me.password', { cur: fm.cur, pass: fm.pass }).then(function (d) { if (d && d.token) { S.token = d.token; store.set('token', d.token); } S.busy = ''; S.err = ''; S.form = {}; toast(T('passChanged')); ACT.backMe(); }).catch(function (er) { S.busy = ''; S.err = errText(er); render(true); });
  }
  if (id === 'codeForm') { var code = $('#code').value.trim(); if (code) { stopCam(); lookup(code); } }
  if (id === 'agForm') {
    var a = S.ag; a.name = a.name || S.me.name; a.phone = a.phone || S.me.phone;
    if (!a.name || !a.phone || !(a.place || '').trim()) { S.err = T('e_empty'); render(true); return; }
    S.busy = T('send'); render(true);
    call('agent.apply', a).then(function () { S.busy = ''; S.ag = { sent: true }; S.err = ''; render(true); }).catch(function (er) { S.busy = ''; S.err = errText(er); render(true); });
  }
});
