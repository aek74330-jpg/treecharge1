function setup() {
  var ss = SpreadsheetApp.getActive();
  ss.setSpreadsheetTimeZone('Asia/Vientiane');
  Object.keys(SCHEMA).forEach(function (name) {
    var s = ss.getSheetByName(name) || ss.insertSheet(name);
    var cols = SCHEMA[name];
    if (s.getMaxColumns() < cols.length) s.insertColumnsAfter(s.getMaxColumns(), cols.length - s.getMaxColumns());
    s.getRange(1, 1, 1, cols.length).setValues([cols]).setFontWeight('bold').setBackground('#0f1a13').setFontColor('#ffffff');
    s.setFrozenRows(1);
    cols.forEach(function (c, i) { if (TEXT_COLS.indexOf(c) >= 0) s.getRange(2, i + 1, s.getMaxRows() - 1, 1).setNumberFormat('@'); });
  });
  var first = ss.getSheetByName('Sheet1') || ss.getSheetByName('ชีต1') || ss.getSheetByName('Feuille 1');
  if (first && ss.getSheets().length > 1 && first.getLastRow() === 0) ss.deleteSheet(first);
  _memo = {};

  for (var k in CONFIG_DEFAULTS) if (!find('Config', k)) insert('Config', { id: k, value: CONFIG_DEFAULTS[k][0], note: CONFIG_DEFAULTS[k][1] });
  if (!config().hookKey) setConfig('hookKey', Utilities.getUuid().replace(/-/g, ''));

  if (!rows('Admins').length) {
    var salt = newSalt_();
    insert('Admins', { id: 'admin', name: 'Admin', role: 'owner', salt: salt, passH: hashPw_(salt, 'tree1234'), created: now_() });
  }
  if (!rows('Branches').length) seedBranches_();
  uploadFolder_();
  try { SpreadsheetApp.getUi().alert('ຕັ້ງຄ່າສຳເລັດ ✓\n\nແອັດມິນ: admin / tree1234 (ປ່ຽນລະຫັດທັນທີໃນໜ້າ ຕັ້ງຄ່າລະບົບ)\n\nຂັ້ນຕໍ່ໄປ: Deploy → New deployment → Web app'); } catch (e) {}
}

/** Remove everything addDemoHistory() made (customers D…, their charges and money, demo franchises). */
function clearDemoData() {
  _memo = {};
  var isDemoCust = function (id) { return /^D\d+$/.test(String(id)); };
  [['Sessions', function (r) { return isDemoCust(r.cust) || r.flags === 'demo'; }], ['Ledger', function (r) { return isDemoCust(r.cust); }],
   ['Customers', function (r) { return isDemoCust(r.id); }], ['Franchises', function (r) { return String(r.id).indexOf('FDEMO') === 0; }],
   ['Tokens', function (r) { return isDemoCust(r.uid) || String(r.uid).indexOf('FDEMO') === 0; }]].forEach(function (x) {
    var s = sheet_(x[0]), cols = SCHEMA[x[0]], keep = rows(x[0]).filter(function (r) { return !x[1](r); });
    if (s.getLastRow() > 1) s.getRange(2, 1, s.getLastRow() - 1, cols.length).clearContent();
    delete _memo[x[0]];
    insertMany(x[0], keep);
  });
  try { SpreadsheetApp.getUi().alert('ລົບຂໍ້ມູນທົດລອງແລ້ວ'); } catch (e) {}
}

/**
 * fixAdmin() — run this when the admin page won't let you in.
 * Puts back the account admin / tree1234 (owner), clears the 15-minute lock after
 * wrong passwords, signs out old admin sessions, and checks the 4 HTML files.
 */
function fixAdmin() {
  _memo = {};
  if (!SpreadsheetApp.getActive().getSheetByName('Admins')) setup();
  _memo = {};
  var salt = newSalt_();
  var cur = rows('Admins').filter(function (x) { return String(x.id).trim().toLowerCase() === 'admin'; })[0];
  var rec = { id: 'admin', name: (cur && cur.name) || 'Admin', role: 'owner', salt: salt, passH: hashPw_(salt, 'tree1234'), created: (cur && cur.created) || now_() };
  if (cur) update('Admins', cur.id, rec); else insert('Admins', rec);
  dropAllTokens_('a', 'admin');
  var c = CacheService.getScriptCache(); ['try:a:admin', 'try:a:Admin', 'try:a:ADMIN', 'try:a:admin '].forEach(function (k) { c.remove(k); });
  var files = ['App', 'AdminUI', 'PartnerUI', 'Logo'].map(function (f) {
    try { HtmlService.createHtmlOutputFromFile(f); return '✓ ' + f; } catch (e) { return '✗ ' + f + '  ← ບໍ່ພົບໄຟລ໌ນີ້ / ไม่พบไฟล์นี้'; }
  });
  var u = ScriptApp.getService().getUrl();
  var msg = 'ແອັດມິນ: admin / tree1234 ✓\n\nໄຟລ໌ HTML:\n' + files.join('\n') +
    '\n\nລິ້ງແອັດມິນ / ลิงก์แอดมิน:\n' + (u ? u + '?page=admin' : '(ຍັງບໍ່ໄດ້ Deploy / ยังไม่ได้ Deploy)');
  console.log(msg);
  try { SpreadsheetApp.getUi().alert(msg); } catch (e) {}
}
